# LLM / orchestration
langchain
# Pinned range: interrupt()/Command resume + the __interrupt__ stream chunk are
# relied on by the HITL flow; verified against 1.2.x.
langgraph>=1.2,<2
langchain-openai
langchain-mcp-adapters
# Durable LangGraph checkpoints (CHECKPOINTER_BACKEND=postgres). AsyncPostgresSaver
# speaks psycopg v3 (NOT asyncpg, which SQLAlchemy keeps using) — two drivers, one DB.
langgraph-checkpoint-postgres
psycopg[binary]
# Local-embeddings extras (langchain-huggingface, sentence-transformers, torch) live in
# requirements-hf.txt, not here — only needed when EMBEDDING_PROVIDER=hf (default is
# "openai" via the LiteLLM proxy). Keeping the ~2-3GB torch download out of the base
# install means CI/Docker builds never touch PyTorch's CDN, which has intermittent SSL
# handshake failures on some networks/runners.

# MCP servers
# Pinned: FastMCP 3.x enforces streamable-HTTP Host/Origin validation (localhost-only by
# default), which 421s the api's calls by Docker service name — handled by
# FASTMCP_HTTP_HOST_ORIGIN_PROTECTION=false in docker-compose.yml. Pin so a rebuild can't
# silently jump versions and change transport behavior again.
fastmcp==3.4.3
PyGithub
